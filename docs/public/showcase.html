MethodSignatureDescription
show(message, options?) → stringShow toast. Options: { type, title, duration }. Returns ID.

const toast = document.querySelector('ai-toast');

// Simple

  type: 'success',
  title: 'Deploy Complete',
  duration: 8000
});

// Persistent (no auto-dismiss)
toast.show('Processing...', { type: 'ai', duration: 0 });

import { OpenAiClient } from '@cognivo/adapter-openai';
const client = new OpenAiClient({ apiKey: '...' });
const chat = document.querySelector('ai-chat');
chat.aiClient = client;
chat.intent = 'explain';
chat.chatDataset = myData;

// Enable streaming
